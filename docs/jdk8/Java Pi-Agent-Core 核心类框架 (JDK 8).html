Agent 执行引擎 (pi-agent-runtime) // ============================================================================ package com.earendil.pi.agent; import com.earendil.pi.session.SessionNode; import com.earendil.pi.session.SessionTree; import com.earendil.pi.session.SessionManager; import com.earendil.pi.tool.ToolRegistry; import com.earendil.pi.llm.LLMAdapter; import com.earendil.pi.llm.Message; import java.util.*; import java.util.concurrent.CompletableFuture; import java.util.concurrent.atomic.AtomicInteger; /** * Agent 执行引擎 - 完整的执行循环 */ public class AgentExecutor { private final SessionManager sessionManager; private final ToolRegistry toolRegistry; private final LLMAdapter llmAdapter; private final ContextManager contextManager; private final AtomicInteger activeRequests = new AtomicInteger(0); public AgentExecutor( SessionManager sessionManager, ToolRegistry toolRegistry, LLMAdapter llmAdapter, ContextManager contextManager) { this.sessionManager = sessionManager; this.toolRegistry = toolRegistry; this.llmAdapter = llmAdapter; this.contextManager = contextManager; } /** * 异步执行 Agent 单个循环 */ public CompletableFuture executeAsync(String sessionId, String userMessage) { activeRequests.incrementAndGet(); return sessionManager.getSessionAsync(sessionId) .thenCompose(session -> addUserMessageAsync(session, userMessage)) .thenCompose(session -> callLLMAsync(session)) .thenCompose(response -> handleToolCallsAsync(response)) .thenCompose(response -> sessionManager.saveSessionAsync(response.getSession()) .thenApply(v -> response)) .whenComplete((result, ex) -> activeRequests.decrementAndGet()); } /** * Step 1: 添加用户消息 */ private CompletableFuture addUserMessageAsync(SessionTree session, String userMessage) { return CompletableFuture.supplyAsync(() -> { SessionNode userNode = SessionNode.createUserMessage(session.getId(), userMessage); session.addNode(userNode); return session; }); } /** * Step 2: 调用 LLM */ private CompletableFuture callLLMAsync(SessionTree session) { return CompletableFuture.supplyAsync(() -> { // 构建上下文（会话历史） List messages = session.getNodes().stream() .map(node -> new Message(node.getRole(), node.getContent())) .collect(java.util.stream.Collectors.toList()); // 应用上下文压缩 messages = contextManager.compressContext(messages); return new AgentResponse(session, messages); }).thenCompose(response -> { Map params = new HashMap<>(); params.put("temperature", 0.7); params.put("max_tokens", 2048); return llmAdapter.callAsync(response.getMessages(), params) .thenApply(llmResponse -> { response.setLLMOutput(llmResponse.getContent()); response.setTokenUsage(llmResponse.getPromptTokens(), llmResponse.getCompletionTokens()); return response; }); }); } /** * Step 3: 处理工具调用 */ private CompletableFuture handleToolCallsAsync(AgentResponse response) { String llmOutput = response.getLLMOutput(); // 解析 LLM 输出中的工具调用（示例） List> toolTasks = new ArrayList<>(); // 简单示例：检查输出中是否包含工具调用标记 if (llmOutput.contains("TOOL:")) { // 提取工具名和参数 String[] parts = llmOutput.split("TOOL:"); for (int i = 1; i < parts.length; i++) { String toolCall = parts[i].split("\n")[0]; String[] toolParts = toolCall.split("\\|"); String toolName = toolParts[0].trim(); CompletableFuture task = toolRegistry.executeAsync(toolName, new HashMap<>()) .thenAccept(result -> { // 记录工具执行结果 response.addToolResult(toolName, result.getOutput()); }); toolTasks.add(task); } } // 等待所有工具执行完成 return CompletableFuture.allOf( toolTasks.toArray(new CompletableFuture[0]) ).thenApply(v -> response); } /** * 获取当前活跃请求数 */ public int getActiveRequests() { return activeRequests.get(); } } /** * Agent 响应 */ public class AgentResponse { private SessionTree session; private List messages; private String llmOutput; private int promptTokens; private int completionTokens; private Map toolResults; public AgentResponse(SessionTree session, List messages) { this.session = session; this.messages = messages; this.toolResults = new HashMap<>(); } public SessionTree getSession() { return session; } public List getMessages() { return messages; } public String getLLMOutput() { return llmOutput; } public void setLLMOutput(String output) { this.llmOutput = output; } public void setTokenUsage(int prompt, int completion) { this.promptTokens = prompt; this.completionTokens = completion; } public int getTotalTokens() { return promptTokens + completionTokens; } public void addToolResult(String toolName, String result) { this.toolResults.put(toolName, result); } } /** * 上下文管理器 - 自适应上下文压缩 */ public class ContextManager { private static final int MAX_TOKENS = 128000; private static final int RESERVE_TOKENS = 4000; /** * 压缩会话上下文 */ public List compressContext(List messages) { int totalTokens = messages.stream() .mapToInt(m -> estimateTokens(m.getContent())) .sum(); if (totalTokens < MAX_TOKENS - RESERVE_TOKENS) { return messages; // 无需压缩 } // 使用策略：保留最近的消息，压缩早期对话 int targetTokens = MAX_TOKENS - RESERVE_TOKENS; List compressed = new ArrayList<>(); int currentTokens = 0; // 从后往前添加消息 for (int i = messages.size() - 1; i >= 0; i--) { Message msg = messages.get(i); int msgTokens = estimateTokens(msg.getContent()); if (currentTokens + msgTokens <= targetTokens) { compressed.add(0, msg); currentTokens += msgTokens; } else if (compressed.isEmpty()) { // 至少保留最后一条 compressed.add(0, msg); break; } else { break; } } return compressed; } /** * 估计 token 数（简单实现） */ private int estimateTokens(String text) { // 简单估算：1 token ≈ 4 个字符 return Math.max(1, text.length() / 4);
