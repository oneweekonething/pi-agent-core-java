} @Override public List compress( List entries, int maxTokens ) { int currentTokens = entries.stream() .mapToInt(ContextEntry::tokenCount) .sum(); if (currentTokens <= maxTokens) { return entries; } // 分层压缩策略 List highPriority = entries.stream() .filter(e -> e.layer().priority >= 4) .toList(); } } // 令牌计数器 public class TokenCounter { private final Tokenizer tokenizer; public TokenCounter() { // 使用 OpenAI 的 Tokenizer this.tokenizer = Tokenizer.getEncoding("cl100k_base"); } public int count(Object value) { String text = serialize(value); // Agent 执行器 public class AgentExecutor { private final LLMProvider llmProvider; private final SessionManager sessionManager; private final ToolRegistry toolRegistry; private final ContextManager contextManager; private final SecurityManager securityManager; private final AgentConfig config; public AgentExecutor( LLMProvider llmProvider, SessionManager sessionManager, ToolRegistry toolRegistry, ContextManager contextManager, SecurityManager securityManager, AgentConfig config ) { this.llmProvider = llmProvider; this.sessionManager = sessionManager; this.toolRegistry = toolRegistry; this.contextManager = contextManager; this.securityManager = securityManager; this.config = config; } public Mono execute( String userInput, String sessionId, PermissionContext permissions ) { return sessionManager.getHistory(sessionId) .flatMap(history -> executeLoop( userInput, sessionId, history, permissions, 0 )); } private Mono executeLoop( String userInput, String sessionId, List history, PermissionContext permissions, int toolCallCount ) { return Mono.defer(() -> { // 1. 组装上下文 Map ctx = contextManager.assembleContext(); // 2. 构建 LLM 消息 List messages = buildLLMMessages(history, ctx, userInput); // 3. 调用 LLM return llmProvider.chatCompletion(messages, config.llmConfig()) // 4. 处理响应 .flatMap(response -> handleLLMResponse( response, sessionId, history, permissions, toolCallCount )) .retryWhen(Retry.backoff(3, Duration.ofSeconds(1))); }); } private Mono handleLLMResponse( ChatCompletionResponse response, String sessionId, List history, PermissionContext permissions, int toolCallCount ) { if (response.toolCalls().isEmpty()) { // 无工具调用，直接返回 return Mono.just(new AgentResponse( response.content(), List.of(), true )); } // 执行工具调用 return Flux.fromIterable(response.toolCalls()) .flatMap(toolCall -> toolRegistry.executeTool(toolCall, null) .onErrorResume(e -> Mono.just( new ToolResult( toolCall.id(), ToolStatus.FAILURE, e.getMessage(), Map.of(), 0L ) )) ) .collectList() .flatMap(toolResults -> { // 检查是否继续循环 if (toolCallCount >= config.maxToolCalls()) { return Mono.just(new AgentResponse( "Max tool calls reached", toolResults, false )); } // 继续执行循环 String continuationInput = formatToolResults(toolResults); return executeLoop( continuationInput, sessionId, history, permissions, toolCallCount + 1 ); }); } private List buildLLMMessages( List history, Map context, String userInput ) { List messages = new ArrayList<>(); // 系统消息 messages.add(new SystemMessage(buildSystemPrompt(context))); // 历史消息 history.forEach(node -> { if (node.message() instanceof SessionMessage.UserMessage um) { messages.add(new UserMessage(um.content(), List.of())); } else if (node.message() instanceof SessionMessage.AssistantMessage am) { messages.add(new AssistantMessage(am.content(), am.toolCalls())); } }); // 当前用户输入 messages.add(new UserMessage(userInput, List.of())); return messages; } private String buildSystemPrompt(Map context) { StringBuilder sb = new StringBuilder(); sb.append("You are a helpful AI assistant.\n\n"); List tools = toolRegistry.listAvailableTools(); if (!tools.isEmpty()) { sb.append("Available tools:\n"); tools.forEach(tool -> sb.append(String.format("- %s: %s\n", tool.name(), tool.description())) ); } return sb.toString(); } private String formatToolResults(List results) { return results.stream() .map(r -> String.format("Tool %s: %s", r.toolCallId(), r.output())) .collect(Collectors.joining("\n")); } } // Agent 配置 public record AgentConfig( LLMConfig llmConfig, int maxToolCalls, int contextTokenLimit, boolean enableLogging ) { public static Builder builder() { return new Builder(); } public static class Builder { private LLMConfig llmConfig = LLMConfig.builder("gpt-4").build(); } } } // Agent 响应 public record AgentResponse( String message, List toolResults, boolean success ) {} // ============================================================================ // PART 5: 安全管理核心 // ============================================================================ package com.earendil.pi.core.security;
